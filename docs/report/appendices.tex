\chapter*{Appendix 1: Repository Layout}
\addcontentsline{toc}{chapter}{APPENDIX 1: REPOSITORY LAYOUT}

The directories referred to in the project documentation are shown below.

\begin{lstlisting}[style=plain]
apps/
  api/
    app/
      main.py          # FastAPI app, router mounting, error handlers
      core/            # settings, database session, errors, logging
      models/          # SQLAlchemy 2.0 models (video, track, identity, ...)
      routers/         # health, videos, jobs, experiments, settings
      schemas/         # request / response schemas
      services/        # ProcessingService, storage, frames, analytics,
                       # embedding projection, Celery app
    alembic/           # database migrations
    tests/             # pytest suite
  web/
    src/
      api/             # typed API client and resource wrappers
      components/      # tracking, videos, landing, ui components
      hooks/           # React Query hooks per resource
      layouts/         # application shell
      pages/           # one file per route
      router/          # route table
ml/
  orchardvision_ml/    # reusable ML library
    detection/         # YOLO and mock detectors
    tracking/          # mock, ByteTrack and TAPNext trackers
    identity/          # embedders, identity dataclass, clustering
    memory/            # prototype memory (quality-aware moving average)
    reid/              # re-identification engine
    motion/            # camera-motion estimators
    evaluation/        # metrics and scenario evaluation
    pipeline/          # library-level pipeline
    camera/            # video sources
packages/
  types/               # shared TypeScript types (@orchardvision/types)
docker/                # Dockerfile and compose stack (api, worker, redis)
docs/                  # this report's LaTeX sources
data/                  # SQLite database, uploads, frame cache
models/                # YOLO and TAPNext checkpoints
scripts/               # evaluation and utility scripts
\end{lstlisting}

\chapter*{Appendix 2: Run Commands}
\addcontentsline{toc}{chapter}{APPENDIX 2: RUN COMMANDS}

\begin{lstlisting}[style=plain]
# backend (development; SQLite at data/orchardvision.db)
cd apps/api && uv run uvicorn app.main:app --reload --port 8001

# frontend (Vite dev server, proxies /api/ to the backend)
cd apps/web && npm run dev

# backend tests
cd apps/api && uv run pytest

# frontend unit tests
cd apps/web && npm run test

# end-to-end tests (starts the API and the dev server)
cd apps/web && npm run test:e2e

# type-check the shared types
cd packages/types && npx tsc --noEmit

# container stack: API + Celery worker + Redis
docker compose -f docker/docker-compose.yml up --build
\end{lstlisting}
